/** * Web App de Drive — Consorcio Tato Joven (versión segura) * ---------------------------------------------------------------- * ANTES DE IMPLEMENTAR: * engranaje (Configuración del proyecto) → Propiedades del script → Agregar propiedad * Nombre: CLAVE_ADMIN Valor: una clave larga que inventes (solo la usa la administración) * * - guardar → solo con la clave de administración * - leer → con la clave: devuelve todo (administración) * → con el código de una unidad: devuelve SOLO los datos de esa unidad * - cualquier otro pedido: no devuelve nada */ var ARCHIVO = 'consorcio_tato_joven.json'; function doPost(e) { var p; try { p = JSON.parse(e.postData.contents); } catch (x) { return salida({ ok: false, error: 'formato' }); } var clave = PropertiesService.getScriptProperties().getProperty('CLAVE_ADMIN'); var admin = !!clave && p.clave === clave; var it = DriveApp.getFoldersByName('Consorcio Tato Joven'); var dir = it.hasNext() ? it.next() : DriveApp.createFolder('Consorcio Tato Joven'); var out = { ok: true }; if (p.accion === 'ping') { out.admin = admin; } else if (p.accion === 'guardar') { if (!admin) return salida({ ok: false, error: 'sin permiso' }); if (!p.datos || !p.datos.meses || !p.datos.propietarios) return salida({ ok: false, error: 'datos invalidos' }); var f = dir.getFilesByName(ARCHIVO); if (f.hasNext()) { var act = f.next(); dir.createFile('respaldo_' + Utilities.formatDate(new Date(), 'GMT-3', 'yyyyMMdd_HHmm') + '.json', act.getBlob().getDataAsString(), 'application/json'); act.setContent(JSON.stringify(p.datos)); out.id = act.getId(); } else { out.id = dir.createFile(ARCHIVO, JSON.stringify(p.datos), 'application/json').getId(); } } else if (p.accion === 'leer') { var g = dir.getFilesByName(ARCHIVO); var datos = g.hasNext() ? JSON.parse(g.next().getBlob().getDataAsString()) : null; if (admin) { out.datos = datos; } else { out.datos = reducir(datos, p.codigo); if (!out.datos) Utilities.sleep(1500); // frena los intentos de adivinar códigos } } else { return salida({ ok: false, error: 'accion' }); } return salida(out); } function reducir(d, codigo) { var c = String(codigo || '').trim().toUpperCase(); if (!d || !c || !d.propietarios) return null; var p = null; for (var i = 0; i < d.propietarios.length; i++) { if (String(d.propietarios[i].codigo || '').toUpperCase() === c) { p = d.propietarios[i]; break; } } if (!p) return null; var cfg = d.config || {}, conf = {}; ['expensaFija', 'expensaAnterior', 'expensaDesde', 'recargoPct', 'vencDia', 'cuotaFondo', 'fondoDesde'].forEach(function (k) { if (cfg[k] !== undefined) conf[k] = cfg[k]; }); var propios = function (x) { return x.uf === p.uf; }; var meses = {}; Object.keys(d.meses || {}).forEach(function (k) { var m = d.meses[k]; meses[k] = { label: m.label, cobros: (m.cobros || []).filter(propios), egresos: [], convenios: (m.convenios || []).filter(propios), recibos: (m.recibos || []).filter(propios) }; }); var f = d.fondoObras || {}; return { mesActivo: d.mesActivo, config: conf, datosPago: d.datosPago || {}, datosPagoFondo: d.datosPagoFondo || {}, propietarios: [{ uf: p.uf, nombre: p.nombre, codigo: p.codigo }], meses: meses, fondoObras: { cobros: (f.cobros || []).filter(propios), gastos: [], rendimientos: [], nextCobroId: 1, nextGastoId: 1, nextRendId: 1 } }; } function salida(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); }
